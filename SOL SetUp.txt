# Step 1: Go to home directory
cd ~

# Step 2: Download micromamba binary
curl -Ls https://micro.mamba.pm/api/micromamba/linux-64/latest | tar -xvj bin/micromamba

# Step 3: Add micromamba to environment paths
echo 'export MAMBA_ROOT_PREFIX=$HOME/micromamba' >> ~/.bashrc
echo 'export PATH=$HOME/bin:$PATH' >> ~/.bashrc
source ~/.bashrc

# Step 4: Initialize micromamba shell
eval "$(micromamba shell hook --shell bash)"

# Step 5: Create Python 3.11 environment
micromamba create -y -n liassist python=3.11

# Step 6: Activate environment
micromamba activate liassist
